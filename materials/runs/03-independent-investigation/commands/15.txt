test('两个 HTTP 编辑者不能用旧修订覆盖已保存正文，重读与 CLI/Store 一致',async()=>{
  const archive=JSON.parse(readFileSync('fixtures/three-chapter-archive.json','utf8'));
  const store=new Store(join(mkdtempSync(join(tmpdir(),'novel-prototype-')),'fixture.sqlite'));
  const id='prototype-'+randomUUID();const work=store.init(workInput.parse({...archive.work,id}));
  const chapters:Chapter[]=[];for(const c of archive.chapters){chapters.push(store.saveDraft(draftInput.parse({workId:id,chapterNumber:c.number,expectedWorkRevision:work.revision,expectedChapterRevision:0,expectedPreviousRevisions:chapters.map(p=>({number:p.number,revision:p.revision})),title:c.title,body:c.body}),'human'));}
  const server=prototypeServer(store,id);server.listen(0,'127.0.0.1');await once(server,'listening');const address=server.address();assert(address&&typeof address!=='string');const base='http://127.0.0.1:'+address.port;
  try{
    const before=await(await fetch(base+'/api/work?work='+id)).json();assert.deepEqual(before.chapters.map((c:{body:string})=>c.body),archive.chapters.map((c:{body:string})=>c.body));
    const input={workId:id,chapterNumber:2,expectedWorkRevision:1,expectedChapterRevision:1,expectedPreviousRevisions:[{number:1,revision:1}],title:chapters[1].title,body:chapters[1].body+'\n\n人工修改：林澈停下来再听一次。'};
    const a=await fetch(base+'/api/draft',{method:'POST',body:JSON.stringify(input)});assert.equal(a.status,200);const saved=await a.json();assert.equal(saved.revision,2);assert.equal(saved.origin,'human');assert.equal(saved.taskId,null);
    const b=await fetch(base+'/api/draft',{method:'POST',body:JSON.stringify({...input,body:'旧窗口的修改不能覆盖'})});assert.equal(b.status,409);assert.equal((await b.json()).error.code,'STALE_CHAPTER');
    const after=await(await fetch(base+'/api/work?work='+id)).json();assert.equal(after.chapters[1].body,input.body);assert.equal(after.chapters[1].revision,2);assert.equal(after.chapters[2].basisCurrent,false);assert.deepEqual(store.getChapter(id,2),saved);
    const invalid=await fetch(base+'/api/draft',{method:'POST',body:JSON.stringify({...input,body:''})});assert.equal(invalid.status,400);assert.equal((await invalid.json()).error.code,'INVALID_INPUT');
  }finally{await new Promise<void>((resolve,reject)=>server.close(error=>error?reject(error):resolve()));store.close();}
});

test('真实保存可重开读取；人的修改令旧提议失效',async()=>{
  const directory=await mkdtemp(resolve(tmpdir(),'agent-lab-state-'));
  const db=resolve(directory,'novel.sqlite');
  const seed=workInput.parse(JSON.parse(await readFile('seed/钟楼.json','utf8')));
  const store=new Store(db);
  store.init(seed);
  const input={workId:seed.id,chapterNumber:1,expectedWorkRevision:1,expectedChapterRevision:0,expectedPreviousRevisions:[],title:'人工测试章节',body:'这段人工文字仅用于验证存储，不是模型生成。'};
  const first=store.saveDraft(input,'agent');
  assert.equal(first.status,'draft');
  store.close();
  const reopened=new Store(db);
  assert.equal(reopened.getChapter(seed.id,1)?.body,input.body);
  const edited=reopened.saveDraft({...input,expectedChapterRevision:1,body:'人修改后的有效正文。'},'human');
  assert.equal(edited.revision,2);
  assert.throws(()=>reopened.saveDraft({...input,expectedChapterRevision:1,body:'旧提议覆盖人的修改。'},'agent'),error=>error instanceof LabError&&error.code==='STALE_CHAPTER');
  assert.equal(reopened.getChapter(seed.id,1)?.body,'人修改后的有效正文。');
  assert.equal(reopened.db.prepare('SELECT COUNT(*) AS count FROM chapter_versions').get()?.count,2);
  reopened.close();
});

test('三章按真实前文修订推进，改稿令旧引用失效，导出保留准确状态与正文',async()=>{
  const store=new Store(':memory:');
  try{
    const first=workInput.parse(JSON.parse(await readFile('seed/钟楼.json','utf8')));
    const three=workInput.parse(JSON.parse(await readFile('seed/钟楼三章.json','utf8')));
    store.init(first);
    const base={workId:first.id,chapterNumber:1,expectedWorkRevision:1,expectedChapterRevision:0,expectedPreviousRevisions:[],title:'测试第一章',body:'人工测试初稿。'};
    store.saveDraft(base,'agent');
    store.updateWork(three,1);
    assert.equal(store.getWorkVersion(first.id,1).facts.length,3);
    assert.equal(store.getWorkVersion(first.id,2).outline.length,3);
    assert.throws(()=>store.saveDraft({...base,expectedChapterRevision:1},'agent'),error=>error instanceof LabError&&error.code==='STALE_WORK');
    store.saveDraft({...base,expectedWorkRevision:2,expectedChapterRevision:1,body:'人工入口改稿示范。'},'human');
    const adopted=store.adopt(first.id,1,2,'测试采用入口，不代表真实读者品质判断。');
    assert.equal(adopted.revision,3);
    assert.equal(adopted.status,'adopted');
    const second={...base,chapterNumber:2,expectedWorkRevision:2,title:'测试第二章',body:'接续人工改稿的测试正文。',expectedPreviousRevisions:[{number:1,revision:2}]};
    assert.throws(()=>store.saveDraft(second,'agent'),error=>error instanceof LabError&&error.code==='STALE_PREVIOUS');
    const third={...second,chapterNumber:3,title:'测试第三章',body:'第三章测试正文。',expectedPreviousRevisions:[{number:1,revision:3},{number:2,revision:1}]};
    assert.throws(()=>store.saveDraft(third,'agent'),error=>error instanceof LabError&&error.code==='PREVIOUS_CHAPTER_MISSING');
    store.saveDraft({...second,expectedPreviousRevisions:[{number:1,revision:3}]},'agent');
    assert.equal(store.exportWork(first.id).allOutlineChaptersSaved,false);
    store.saveDraft(third,'agent');
    const exported=store.exportWork(first.id);
    assert.equal(exported.allOutlineChaptersSaved,true);
    assert.deepEqual(exported.chapters.map(chapter=>[chapter.number,chapter.revision,chapter.status]),[[1,3,'adopted'],[2,1,'draft'],[3,1,'draft']]);
    assert.equal(exported.chapters.every(chapter=>chapter.basisCurrent),true);
    assert.equal(exported.chapters[0].reviewNote,adopted.reviewNote);
    assert.ok(exported.text.includes(`采用说明：${adopted.reviewNote}`));
    for(const body of ['人工入口改稿示范。',second.body,third.body])assert.ok(exported.text.includes(body));
    store.saveDraft({...base,expectedWorkRevision:2,expectedChapterRevision:3,body:'后来再次修改第一章。'},'human');
    assert.equal(store.exportWork(first.id).chapters[1].basisCurrent,false);
    assert.equal(store.exportWork(first.id).chapters[2].basisCurrent,false);
  }finally{store.close();}
});
